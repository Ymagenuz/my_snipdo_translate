# The only setting to change; the path may contain spaces.
$exePath = "C:\Tools\SnipDoTranslate\SnipDoTranslate.exe"

$inputText = [string]::Join(" ", [string[]]$args)
$exeAvailable = Test-Path -LiteralPath $exePath -PathType Leaf

if ([string]::IsNullOrWhiteSpace($inputText)) {
    if (!$exeAvailable) {
        exit 1
    }
    try {
        Start-Process -FilePath $exePath -ArgumentList @("--show") -ErrorAction Stop | Out-Null
        exit 0
    } catch {
        exit 2
    }
}

$tempFile = Join-Path (
    [System.IO.Path]::GetTempPath()
) ("SnipDoTranslate-" + [guid]::NewGuid().ToString("N") + ".txt")

try {
    $utf8WithoutBom = [System.Text.UTF8Encoding]::new($false)
    [System.IO.File]::WriteAllText($tempFile, $inputText, $utf8WithoutBom)
} catch {
    # If writing fails, don't try to delete a temporary file that may exist.
    exit 3
}

if (!$exeAvailable) {
    # The selection is safely in the temporary file; keep it even if the path is wrong.
    exit 1
}

try {
    Start-Process -FilePath $exePath -ArgumentList @(
        "--file",
        "`"$tempFile`"",
        "--delete-after"
    ) -ErrorAction Stop | Out-Null
} catch {
    # Keep the temporary file if the EXE fails to start; once it starts, only an accepted ACK deletes it.
    exit 4
}

exit 0
